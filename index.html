var SHEET_NAME_RESULTS = 'Sheet1'; // ชื่อแท็บเก็บคะแนน var SHEET_ID_EMPLOYEES = '10OZhC1HJ3nmX0enyIbELkUhN79ODA46gTZ4uqPwBeLU'; // ID ไฟล์รายชื่อ var SHEET_NAME_EMPLOYEES = 'Sheet1'; // ชื่อแท็บไฟล์รายชื่อ function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var data = JSON.parse(e.postData.contents); var sheetResults = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_RESULTS); // 1. ตรวจสอบชื่อพนักงานก่อนเข้าสอบ if (data.action === "checkStudent") { var searchEn = String(data.studentId).trim().replace(/^0+/, ''); var empSheet = SpreadsheetApp.openById(SHEET_ID_EMPLOYEES).getSheetByName(SHEET_NAME_EMPLOYEES); if(empSheet) { var empData = empSheet.getDataRange().getValues(); var colEN = -1, colName = -1, colDept = -1; // ค้นหาคอลัมน์จาก 3 แถวแรก (เผื่อมีเว้นบรรทัดด้านบน) for(var r=0; r<3; r++) { if(empData[r]) { var tempEn = empData[r].findIndex(function(h) { return String(h).toUpperCase().includes("EN"); }); if(tempEn > -1) { colEN = tempEn; colName = empData[r].findIndex(function(h) { return String(h).toUpperCase().includes("NAME") || String(h).includes("ชื่อ"); }); colDept = empData[r].findIndex(function(h) { return String(h).toUpperCase().includes("DEPT") || String(h).includes("แผนก"); }); break; } } } if(colEN > -1) { for(var i=1; i -1 ? empData[i][colName] : "ไม่ระบุชื่อ", dept: colDept > -1 ? empData[i][colDept] : "-" })).setMimeType(ContentService.MimeType.JSON); } } } } return ContentService.createTextOutput(JSON.stringify({ found: false })).setMimeType(ContentService.MimeType.JSON); } // 2. ระบบตั้งค่าเปิด/ปิดระบบของครู if (data.action === "setSystemStatus") { sheetResults.getRange("BA1").setValue(data.isOpen); return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON); } // 3. ระบบบันทึกคำตอบของพนักงาน (ค้นหาแถวเดิม) var en = String(data.studentId).trim().replace(/^0+/, ''); var dataRange = sheetResults.getDataRange().getValues(); var rowIndex = -1; for(var i = dataRange.length - 1; i > 0; i--) { var rowEn = String(dataRange[i][1]).trim().replace(/^0+/, ''); if(rowEn === en) { rowIndex = i + 1; break; } } var passFail = ""; if (data.status === "submitted") passFail = data.score >= 80 ? "Pass" : "Fail"; var rowData = [ new Date(), data.studentId, data.name, data.dept, data.status, data.firstScore !== undefined && data.firstScore !== null ? data.firstScore : "", data.score !== undefined && data.score !== null ? data.score : "", passFail ]; for (var j = 0; j < 40; j++) { var ans = (data.answers && data.answers.length > j) ? data.answers[j] : null; rowData.push(ans === null || ans === "" ? "" : ans); } if (rowIndex > -1) { sheetResults.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]); } else { sheetResults.appendRow(rowData); } return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({error: err.toString()})).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doGet(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_RESULTS); var data = sheet.getDataRange().getValues(); var students = []; var sysOpen = true; try { var val = sheet.getRange("BA1").getValue(); if(val === false || val === "false" || val === "FALSE") sysOpen = false; } catch(ex) {} // ข้ามแถวที่ 1 (หัวตาราง) เริ่มดึงข้อมูลจากแถวที่ 2 for(var i = 1; i < data.length; i++) { var row = data[i]; // ถ้าไม่มีรหัส EN ให้ข้ามแถวนี้ไปเลย if(!row[1] || row[1] === "") continue; var answers = []; // ดึงคำตอบข้อ 1-40 (เริ่มที่ index 8) for(var j = 8; j < 48; j++) { // เช็คว่า row มีความยาวถึงคอลัมน์ j หรือไม่ (ป้องกัน Error กรณีเพิ่งเริ่มสอบ) if (j < row.length && row[j] !== "") { answers.push(parseInt(row[j], 10)); } else { answers.push(null); } } // เอาเครื่องหมาย ' ออก (ถ้ามี) var cleanEn = String(row[1]).replace(/'/g, '').trim(); students.push({ studentId: cleanEn, name: row[2] ? String(row[2]) : "", dept: row[3] ? String(row[3]) : "", status: row[4] ? String(row[4]) : "", firstScore: (row[5] === "" || row[5] == null) ? undefined : parseFloat(row[5]), score: (row[6] === "" || row[6] == null) ? null : parseFloat(row[6]), result: row[7] ? String(row[7]) : "", answers: answers }); } // โชว์ข้อมูลให้หน้าเว็บผู้คุมสอบ return ContentService.createTextOutput(JSON.stringify({ systemIsOpen: sysOpen, students: students })).setMimeType(ContentService.MimeType.JSON).setHeader("Access-Control-Allow-Origin", "*"); } catch(err) { // ถ้า Error ให้ส่ง Error กลับมา (แต่ตารางจะไม่ว่างเปล่าแบบไร้สาเหตุ) return ContentService.createTextOutput(JSON.stringify({ error: err.toString(), line: err.lineNumber || "Unknown" })).setMimeType(ContentService.MimeType.JSON).setHeader("Access-Control-Allow-Origin", "*"); } } function doOptions(e) { return ContentService.createTextOutput("").setMimeType(ContentService.MimeType.TEXT).setHeader("Access-Control-Allow-Origin", "*").setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS").setHeader("Access-Control-Allow-Headers", "Content-Type"); }
